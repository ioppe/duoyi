import 'dart:convert';
import 'dart:io';
import 'dart:ui' as ui;

import 'package:duoyi/core/app_typography.dart';
import 'package:duoyi/models/pomodoro.dart';
import 'package:duoyi/core/design_tokens.dart';
import 'package:duoyi/providers/auth_provider.dart';
import 'package:duoyi/providers/custom_focus_sound_provider.dart';
import 'package:duoyi/providers/focus_room_provider.dart';
import 'package:duoyi/providers/pomodoro_provider.dart';
import 'package:duoyi/providers/theme_provider.dart';
import 'package:duoyi/screens/pomodoro_screen.dart';
import 'package:duoyi/widgets/focus_rest_settings.dart';
import 'package:flutter/foundation.dart';
import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';
import 'package:flutter/services.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:provider/provider.dart';
import 'package:shared_preferences/shared_preferences.dart';

void main() {
  testWidgets('capture real Flutter focus/rest layouts and modal geometry', (tester) async {
    final output = Directory(Platform.environment['FOCUS_REST_SCREENSHOTS']!);
    await tester.runAsync(() async {
      await output.create(recursive: true);
      final font = FontLoader(DesignTokens.fontFamily)
        ..addFont(rootBundle.load('assets/fonts/NotoSansSC-Variable.ttf'));
      await font.load();
      final icons = FontLoader('MaterialIcons')
        ..addFont(rootBundle.load('fonts/MaterialIcons-Regular.otf'));
      await icons.load();
    });
    final geometry = <String, Object?>{};
    final boundaryKey = GlobalKey();
    PomodoroProvider? current;

    Future<void> screen(Size size, TargetPlatform platform, {double scale = 1, int skips = 0}) async {
      await tester.pumpWidget(const SizedBox.shrink());
      await tester.pumpAndSettle();
      current?.dispose();
      SharedPreferences.setMockInitialValues(<String, Object>{});
      debugDefaultTargetPlatformOverride = platform;
      tester.view.physicalSize = size;
      tester.view.devicePixelRatio = 1;
      tester.view.viewInsets = const FakeViewPadding();
      final provider = PomodoroProvider();
      current = provider;
      for (var i = 0; i < skips; i++) {
        if (provider.state.type == PomodoroType.focus) {
          provider.finishCurrentSession();
        } else {
          provider.skipSession();
        }
      }
      final theme = ThemeProvider();
      await tester.pumpWidget(
        MultiProvider(
          providers: [
            ChangeNotifierProvider<PomodoroProvider>.value(value: provider),
            ChangeNotifierProvider<ThemeProvider>(create: (_) => theme),
            ChangeNotifierProvider<AuthProvider>(create: (_) => AuthProvider()),
            ChangeNotifierProvider<FocusRoomProvider>(create: (_) => FocusRoomProvider()),
            ChangeNotifierProvider<CustomFocusSoundProvider>(create: (_) => CustomFocusSoundProvider()),
          ],
          child: RepaintBoundary(
            key: boundaryKey,
            child: MaterialApp(
              debugShowCheckedModeBanner: false,
              theme: theme.brand.theme,
              builder: (context, child) => MediaQuery(
                data: MediaQuery.of(context).copyWith(
                  textScaler: AppTypography.platformScaler(system: TextScaler.linear(scale)),
                ),
                child: child!,
              ),
              home: const PomodoroScreen(),
            ),
          ),
        ),
      );
      await tester.pumpAndSettle();
      expect(tester.takeException(), isNull);
    }

    Map<String, num> rect(Rect value) => {
      'left': value.left, 'top': value.top,
      'width': value.width, 'height': value.height,
      'bottom': value.bottom,
    };

    Future<void> capture(String name) async {
      await tester.pumpAndSettle();
      expect(tester.takeException(), isNull, reason: name);
      final boundary = boundaryKey.currentContext!.findRenderObject()! as RenderRepaintBoundary;
      await tester.runAsync(() async {
        final image = await boundary.toImage(pixelRatio: 2);
        final bytes = await image.toByteData(format: ui.ImageByteFormat.png);
        await File('${output.path}/$name.png').writeAsBytes(bytes!.buffer.asUint8List());
        image.dispose();
      });
    }

    Future<void> open() async {
      await tester.tap(find.byKey(FocusRestKeys.summary));
      await tester.pumpAndSettle();
      expect(tester.takeException(), isNull);
      expect(find.byKey(FocusRestKeys.save).hitTestable(), findsOneWidget);
    }

    await screen(const Size(390, 844), TargetPlatform.android, scale: 1.25);
    geometry['phoneSummary'] = rect(tester.getRect(find.byKey(FocusRestKeys.summary)));
    geometry['phonePrimary'] = rect(tester.getRect(find.widgetWithText(FilledButton, '开始专注')));
    await capture('phone-main');
    await open();
    final saveBefore = tester.getRect(find.byKey(FocusRestKeys.save));
    geometry['phoneSettings'] = rect(tester.getRect(find.byKey(FocusRestKeys.sheet)));
    geometry['phoneSave'] = rect(saveBefore);
    await capture('phone-settings');
    await tester.drag(find.descendant(of: find.byKey(FocusRestKeys.sheet), matching: find.byType(ListView)), const Offset(0, -550));
    await tester.pumpAndSettle();
    expect(tester.getRect(find.byKey(FocusRestKeys.save)).center.dy, closeTo(saveBefore.center.dy, 0.1));
    expect(find.byKey(FocusRestKeys.autoFocusSwitch).hitTestable(), findsOneWidget);
    await capture('phone-settings-scrolled');

    await screen(const Size(390, 844), TargetPlatform.android, scale: 1.25, skips: 1);
    await capture('phone-short-break');
    await screen(const Size(390, 844), TargetPlatform.android, scale: 1.25, skips: 7);
    await capture('phone-long-break');

    await screen(const Size(320, 720), TargetPlatform.android, scale: 1.5);
    expect(find.text('开始专注').hitTestable(), findsOneWidget);
    await capture('phone-320-large-text');
    await open();
    await capture('phone-320-settings-large-text');
    tester.view.viewInsets = const FakeViewPadding(bottom: 300);
    await tester.pumpAndSettle();
    expect(tester.takeException(), isNull);
    final keyboardSave = tester.getRect(find.byKey(FocusRestKeys.save));
    expect(keyboardSave.bottom, lessThanOrEqualTo(420));
    expect(find.byKey(FocusRestKeys.save).hitTestable(), findsOneWidget);
    geometry['keyboardSave'] = rect(keyboardSave);
    await capture('phone-320-settings-keyboard');

    await screen(const Size(1280, 800), TargetPlatform.macOS);
    geometry['desktopSummary'] = rect(tester.getRect(find.byKey(FocusRestKeys.summary)));
    geometry['desktopPrimary'] = rect(tester.getRect(find.widgetWithText(FilledButton, '开始专注')));
    await capture('desktop-main');
    await open();
    final modal = tester.getRect(find.byKey(FocusRestKeys.sheet));
    expect(modal.width, lessThanOrEqualTo(464));
    expect(modal.center.dx, closeTo(640, 1));
    geometry['desktopModal'] = rect(modal);
    await capture('desktop-settings');

    await tester.pumpWidget(const SizedBox.shrink());
    await tester.pumpAndSettle();
    current?.dispose();
    tester.view.reset();
    debugDefaultTargetPlatformOverride = null;
    await tester.runAsync(() => File('${output.path}/geometry.json').writeAsString(const JsonEncoder.withIndent('  ').convert(geometry)));
  });
}
